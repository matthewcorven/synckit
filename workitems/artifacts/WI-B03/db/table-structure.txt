-- SQL Server verification queries (run against the target database)

EXEC sp_help 'Trials';

SELECT i.name AS index_name, i.is_unique, ic.column_id, c.name AS column_name
FROM sys.indexes i
JOIN sys.index_columns ic ON i.object_id = ic.object_id AND i.index_id = ic.index_id
JOIN sys.columns c ON ic.object_id = c.object_id AND ic.column_id = c.column_id
WHERE i.object_id = OBJECT_ID('Trials')
ORDER BY i.name, ic.column_id;

SELECT name, is_computed, definition
FROM sys.computed_columns
WHERE object_id = OBJECT_ID('Trials');
